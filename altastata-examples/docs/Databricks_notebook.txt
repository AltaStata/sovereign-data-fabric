// RUN SCALA SPARK

val textFile = sparkBuilder.sparkContext.textFile("altastata:///Applications/britannica.txt", 10)

// Split the lines into words and filter out empty words
val words = textFile.flatMap(line => line.split("\\W+")).filter(_.nonEmpty)

// Extract the starting letter of each word and convert to lowercase
val wordsWithLetters = words.map(word => word.toLowerCase.charAt(0).toString)

// Convert to DataFrame and group by letter
import spark.implicits._
val letterCounts = wordsWithLetters.toDF("letter")
  .groupBy("letter")
  .count()
  .orderBy("letter")

letterCounts.createOrReplaceTempView("firstLetters")
